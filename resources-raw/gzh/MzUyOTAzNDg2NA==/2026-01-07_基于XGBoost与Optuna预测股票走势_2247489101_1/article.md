本文目标是采用系统化的机器学习流程构建A股量化交易策略。

  

首先通过Akshare获取股票历史数据，随后基于价格和成交量构建多维技术指标特征体系，涵盖趋势、动量、波动率等维度。

  

通过定义未来涨跌作为监督学习标签，采用XGBoost算法结合Optuna进行超参数优化，以最大化策略收益为目标进行模型训练。

  

最终通过严谨的前向验证框架进行策略回测，评估模型在历史数据上的表现，并与买入持有基准进行对比，形成完整的从数据处理到策略评估的量化交易闭环。

def calculate_technical_indicators(df):    """    计算技术指标特征    """    if df.empty:        return df        df = df.copy()        # 确保日期索引    if'date'in df.columns:        df = df.set_index('date').sort_index()        # 基础价格特征    df['returns'] = df['close'].pct_change()    df['log_returns'] = np.log(df['close'] / df['close'].shift(1))        # 移动平均线    for window in [5, 10, 20, 30, 60]:        df[f'sma_{window}'] = df['close'].rolling(window=window).mean()        df[f'ema_{window}'] = df['close'].ewm(span=window, adjust=False).mean()        # 价格与均线关系    df['close_vs_sma20'] = df['close'] / df['sma_20']    df['close_vs_sma60'] = df['close'] / df['sma_60']    df['sma20_vs_sma60'] = df['sma_20'] / df['sma_60']        # 布林带    for window in [20]:        df[f'bb_middle_{window}'] = df['close'].rolling(window=window).mean()        bb_std = df['close'].rolling(window=window).std()        df[f'bb_upper_{window}'] = df[f'bb_middle_{window}'] + 2 * bb_std        df[f'bb_lower_{window}'] = df[f'bb_middle_{window}'] - 2 * bb_std        df[f'bb_width_{window}'] = (df[f'bb_upper_{window}'] - df[f'bb_lower_{window}']) / df[f'bb_middle_{window}']        df[f'bb_position_{window}'] = (df['close'] - df[f'bb_lower_{window}']) / (df[f'bb_upper_{window}'] - df[f'bb_lower_{window}'])        # RSI相对强弱指标    for window in [6, 12, 24]:        delta = df['close'].diff()        gain = (delta.where(delta > 0, 0)).rolling(window=window).mean()        loss = (-delta.where(delta < 0, 0)).rolling(window=window).mean()        rs = gain / loss        df[f'rsi_{window}'] = 100 - (100 / (1 + rs))        # MACD指标    exp1 = df['close'].ewm(span=12, adjust=False).mean()    exp2 = df['close'].ewm(span=26, adjust=False).mean()    df['macd'] = exp1 - exp2    df['macd_signal'] = df['macd'].ewm(span=9, adjust=False).mean()    df['macd_hist'] = df['macd'] - df['macd_signal']        # 成交量指标    df['volume_sma_20'] = df['volume'].rolling(window=20).mean()    df['volume_ratio'] = df['volume'] / df['volume_sma_20']    df['obv'] = (np.sign(df['close'].diff()) * df['volume']).fillna(0).cumsum()        # 波动率指标    df['atr'] = calculate_atr(df, window=14)    df['atr_pct'] = df['atr'] / df['close']        # 价格通道    for window in [20]:        df[f'high_{window}'] = df['high'].rolling(window=window).max()        df[f'low_{window}'] = df['low'].rolling(window=window).min()        df[f'channel_position_{window}'] = (df['close'] - df[f'low_{window}']) / (df[f'high_{window}'] - df[f'low_{window}'])        # 动量指标    for window in [5, 10, 20]:        df[f'momentum_{window}'] = df['close'] / df['close'].shift(window) - 1        # 收益率相关    df['returns_5'] = df['close'].pct_change(5)    df['returns_10'] = df['close'].pct_change(10)    df['returns_20'] = df['close'].pct_change(20)        # 价格位置特征    df['high_low_ratio'] = (df['high'] - df['low']) / df['close']    df['close_open_ratio'] = (df['close'] - df['open']) / df['open']        # 量价关系    df['vwap'] = (df['amount'] / df['volume']).replace([np.inf, -np.inf], np.nan)    df['price_vs_vwap'] = df['close'] / df['vwap']        # 换手率特征    if'turnover'in df.columns:        df['turnover_sma_20'] = df['turnover'].rolling(window=20).mean()        df['turnover_ratio'] = df['turnover'] / df['turnover_sma_20']        # 振幅特征    if'amplitude'in df.columns:        df['amplitude_sma_20'] = df['amplitude'].rolling(window=20).mean()        df['amplitude_ratio'] = df['amplitude'] / df['amplitude_sma_20']        # 日历特征    if isinstance(df.index, pd.DatetimeIndex):        df['day_of_week'] = df.index.dayofweek        df['month'] = df.index.month        df['quarter'] = df.index.quarter        df['year'] = df.index.year                # 周期性编码        df['day_sin'] = np.sin(2 * np.pi * df.index.dayofyear / 365.25)        df['day_cos'] = np.cos(2 * np.pi * df.index.dayofyear / 365.25)        df['month_sin'] = np.sin(2 * np.pi * df.index.month / 12)        df['month_cos'] = np.cos(2 * np.pi * df.index.month / 12)        # 清理NaN值    df = df.replace([np.inf, -np.inf], np.nan)        return dfdef calculate_atr(df, window=14):    """计算平均真实波幅"""    high = df['high']    low = df['low']    close = df['close'].shift(1)        tr1 = high - low    tr2 = (high - close).abs()    tr3 = (low - close).abs()        tr = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)    atr = tr.rolling(window=window).mean()        return atr# 计算技术指标features_df = calculate_technical_indicators(stock_data)print(f"特征数据集形状: {features_df.shape}")print(f"特征列数: {len(features_df.columns)}")print("\n前5行数据:")print(features_df.head())

class BacktestEngine:    """回测引擎"""        def __init__(self, initial_capital=100000, commission_rate=0.0003):        self.initial_capital = initial_capital        self.commission_rate = commission_rate            def calculate_metrics(self, equity_curve):        """计算策略指标"""        if len(equity_curve) < 2:            return {}                returns = equity_curve.pct_change().dropna()        total_return = equity_curve.iloc[-1] / equity_curve.iloc[0] - 1                # 年化收益率        years = len(equity_curve) / 252        cagr = (1 + total_return) ** (1 / years) - 1if years > 0else0                # 夏普比率        sharpe = np.sqrt(252) * returns.mean() / returns.std() if returns.std() > 0else0                # 最大回撤        cummax = equity_curve.cummax()        drawdown = (equity_curve - cummax) / cummax        max_drawdown = drawdown.min()                # 卡玛比率        calmar = cagr / abs(max_drawdown) if max_drawdown != 0else0                # 胜率        positive_days = (returns > 0).sum()        win_rate = positive_days / len(returns) if len(returns) > 0else0                return {            '总收益率': total_return,            '年化收益率': cagr,            '夏普比率': sharpe,            '最大回撤': max_drawdown,            '卡玛比率': calmar,            '胜率': win_rate,            '年化波动率': returns.std() * np.sqrt(252)        }        def backtest_simple(self, df, signal_col='signal'):        """        简单回测：信号为1时全仓买入，信号为0时卖出        """        df = df.copy()        equity = [self.initial_capital]        position = 0        shares = 0        trades = []                for i in range(1, len(df)):            current_price = df['close'].iloc[i]            prev_signal = df[signal_col].iloc[i-1]            current_signal = df[signal_col].iloc[i]                        # 开仓            if position == 0and prev_signal == 1:                shares = equity[-1] / current_price                position = 1                trades.append({                    'date': df.index[i],                    'action': 'buy',                    'price': current_price,                    'shares': shares,                    'value': equity[-1]                })                        # 平仓            elif position == 1and prev_signal == 0:                trade_value = shares * current_price                trade_value *= (1 - self.commission_rate)  # 扣除佣金                equity[-1] = trade_value                position = 0                shares = 0                trades.append({                    'date': df.index[i],                    'action': 'sell',                    'price': current_price,                    'shares': shares,                    'value': trade_value                })                        # 更新权益            if position == 1:                current_value = shares * current_price            else:                current_value = equity[-1]                        equity.append(current_value)                equity_curve = pd.Series(equity, index=df.index[:len(equity)])        trades_df = pd.DataFrame(trades) if trades else pd.DataFrame()                return equity_curve, trades_df        def backtest_buy_hold(self, df):        """买入持有策略"""        df = df.copy()        shares = self.initial_capital / df['close'].iloc[0]        equity_curve = shares * df['close']        return equity_curve# 初始化回测引擎backtester = BacktestEngine(initial_capital=100000)

y_val=None, params=None):        """训练模型"""        if params isNoneand self.best_params isnotNone:            params = self.best_params                if params isNone:            # 默认参数   'random_state': self.random_state,                'n_jobs': -1            }                # 添加回调        callbacks = []        if X_val isnotNoneand y_val isnotNone:            eval_set = [(X_val, y_val)]            callbacks = [xgb.callback.EarlyStopping(rounds=50)]        else:            eval_set = None                # 训练模型        self.model = xgb.XGBClassifier(**params)        self.model.fit(            X_train, y_train,            eval_set=eval_set,            verbose=False,            callbacks=callbacks        )                # 特征重要性        self.feature_importance = pd.DataFrame({            'feature': X_train.columns,            'importance': self.model.feature_importances_        }).sort_values('importance', ascending=False)                return self.model        def predict(self, X, threshold=0.5):        """预测"""        if self.model isNone:            raise ValueError("模型未训练")                y_pred_proba = self.model.predict_proba(X)[:, 1]        y_pred = (y_pred_proba >= threshold).astype(int)                return y_pred, y_pred_proba        def walk_forward_validation(self, X, y, train_size=0.7, step_size=20, optimize=True):        """前向验证"""        n_samples = len(X)        train_end = int(n_samples * train_size)                all_predictions = []        all_probas = []        all_indices = []        models_info = []                for i in range(train_end, n_samples, step_size):            # 训练集            X_train = X.iloc[:i]            y_train = y.iloc[:i]                        # 测试集            test_start = i            test_end = min(i + step_size, n_samples)            X_test = X.iloc[test_start:test_end]            y_test = y.iloc[test_start:test_end]                        if len(X_test) == 0:                continue                        # 划分验证集            val_size = int(0.2 * len(X_train))            X_train_final = X_train.iloc[:-val_size]            y_train_final = y_train.iloc[:-val_size]            X_val = X_train.iloc[-val_size:]            y_val = y_train.iloc[-val_size:]                        if optimize:                # 优化参数                best_params = self.optimize_hyperparameters(                    X_train_final, y_train_final, X_val, y_val, n_trials=50                )            else:                best_params = None                        # 训练模型            self.train(X_train_final, y_train_final, X_val, y_val, best_params)                        # 预测            y_pred, y_pred_proba = self.predict(X_test)                        # 保存结果            all_predictions.extend(y_pred)            all_probas.extend(y_pred_proba)            all_indices.extend(X_test.index.tolist())                        # 保存模型信息            models_info.append({                'train_end': i,                'test_start': test_start,                'test_end': test_end,                'train_samples': len(X_train_final),                'val_samples': len(X_val),                'test_samples': len(X_test)            })                # 创建结果DataFrame        results_df = pd.DataFrame({            'prediction': all_predictions,            'probability': all_probas        }, index=all_indices)                return results_df, models_info# 创建预测器predictor = StockPredictor(random_state=42)# 准备数据X, y = predictor.prepare_features(labeled_df)#
